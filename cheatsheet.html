3Sum pattern
Arrays.sort(nums);
for (int i = 0; i < nums.length - 2; i++) {
    if (i > 0 && nums[i] == nums[i-1]) continue; // skip dup
    int left = i+1, right = nums.length-1;
    while (left < right) {
        int sum = nums[i] + nums[left] + nums[right];
        if (sum == 0) { /* add, move both, skip dups */ }
        else if (sum < 0) left++;
        else right--;
    }
}

03 Sliding Window 4 / 8

FIXED SIZE WINDOW

Template
int sum = 0;
// build first window
for (int i = 0; i < k; i++) sum += arr[i];
int max = sum;

// slide: add right, remove left
for (int i = k; i < n; i++) {
    sum += arr[i] - arr[i - k];
    max = Math.max(max, sum);
}
return max;

VARIABLE SIZE WINDOW

Expand right, shrink left when invalid
int left = 0, result = 0;
Map<Character, Integer> window = new HashMap<>();

for (int right = 0; right < s.length(); right++) {
    char c = s.charAt(right);
    window.merge(c, 1, Integer::sum);   // add to window

    while (window.get(c) > 1) {         // invalid?
        char lc = s.charAt(left++);
        window.merge(lc, -1, Integer::sum);
        if (window.get(lc) == 0)
            window.remove(lc);
    }
    result = Math.max(result, right - left + 1);
}
return result;
Window size = right - left + 1. Always.

CHARACTER FREQUENCY WINDOW

Using int[26] instead of HashMap (faster)
int[] freq = new int[26];
for (char c : p.toCharArray()) freq[c - 'a']++;

int[] window = new int[26];
int left = 0;
for (int right = 0; right < s.length(); right++) {
    window[s.charAt(right) - 'a']++;
    if (right - left + 1 > p.length())
        window[s.charAt(left++) - 'a']--;
    if (Arrays.equals(window, freq))
        result.add(left);
}
c - 'a' converts char to index 0-25. Use this for lowercase letter problems.

05 Binary Search 4 / 14

CLASSIC BINARY SEARCH

Exact target
int left = 0, right = arr.length - 1;
while (left <= right) {
    int mid = left + (right - left) / 2; // overflow-safe!
    if (arr[mid] == target) return mid;
    else if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
}
return -1;
⚠ Never use (left + right) / 2 — it overflows for large indices. Always use left + (right - left) / 2.

BOUNDARY BINARY SEARCH

Find leftmost valid (left < right)
int left = lo, right = hi;
while (left < right) {
    int mid = left + (right - left) / 2;
    if (feasible(mid)) right = mid;
    else left = mid + 1;
}
return left;
Search in rotated array
while (left <= right) {
    int mid = left + (right - left) / 2;
    if (nums[mid] == target) return mid;

    // left half is sorted
    if (nums[left] <= nums[mid]) {
        if (nums[left] <= target && target < nums[mid])
            right = mid - 1;
        else left = mid + 1;
    } else { // right half is sorted
        if (nums[mid] < target && target <= nums[right])
            left = mid + 1;
        else right = mid - 1;
    }
}

DFS TRAVERSALS

TreeNode definition
class TreeNode {
    int val; TreeNode left, right;
    TreeNode(int val) { this.val = val; }

All three orders
// Preorder:  root → left → right
// Inorder:   left → root → right  (BST = sorted!)
// Postorder: left → right → root

void dfs(TreeNode node) {
    if (node == null) return;  // base case FIRST
    // preorder: process(node.val)
    dfs(node.left);
    // inorder: process(node.val)
    dfs(node.right);
    // postorder: process(node.val)
}
Max depth pattern
int maxDepth(TreeNode root) {
    if (root == null) return 0;
    return 1 + Math.max(maxDepth(root.left),
                         maxDepth(root.right));
}

BFS — LEVEL ORDER

Template (use Queue, not Deque)
Queue<TreeNode> q = new LinkedList<>();
if (root != null) q.offer(root);
List<List<Integer>> result = new ArrayList<>();

while (!q.isEmpty()) {
    int size = q.size();          // snapshot level size
    List<Integer> level = new ArrayList<>();
    for (int i = 0; i < size; i++) {
        TreeNode node = q.poll();
        level.add(node.val);
        if (node.left != null) q.offer(node.left);
        if (node.right != null) q.offer(node.right);

    result.add(level);

BFS right side view: just take the last element of each level.

BST OPERATIONS

BST property: inorder = sorted ascending
// Validate BST
boolean valid(TreeNode node, long min, long max) {
    if (node == null) return true;
    if (node.val <= min || node.val >= max) return false;
    return valid(node.left, min, node.val)
        && valid(node.right, node.val, max);
}
// call: valid(root, Long.MIN_VALUE, Long.MAX_VALUE)
BST search
TreeNode search(TreeNode root, int target) {
    if (root == null || root.val == target) return root;
    if (target < root.val) return search(root.left, target);
    return search(root.right, target);
}
⚠ Use Long.MIN_VALUE / Long.MAX_VALUE for BST bounds to handle Integer edge cases.

09 Backtracking 2 / 17

UNIVERSAL TEMPLATE

Choose → Explore → Undo
List<List<Integer>> result = new ArrayList<>();

void backtrack(int start, List<Integer> curr) {
    result.add(new ArrayList<>(curr)); // snapshot!

    for (int i = start; i < nums.length; i++) {
        curr.add(nums[i]);               // choose
        backtrack(i + 1, curr);         // explore
        curr.remove(curr.size()-1);    // undo
    }
}
With duplicates (sort + skip)
Arrays.sort(nums); // must sort first

for (int i = start; i < nums.length; i++) {
    if (i > start && nums[i] == nums[i-1]) continue; // skip dup
    curr.add(nums[i]);
    backtrack(i + 1, curr);
    curr.remove(curr.size()-1);
}
⚠ new ArrayList<>(curr) — without this you store a reference. The list will be empty at the end.

void backtrack(List<Integer> curr) {
    if (curr.size() == nums.length) { result.add(new ArrayList<>(curr)); return; }
    for (int i = 0; i < nums.length; i++) {
        if (used[i]) continue;
        used[i] = true;
        curr.add(nums[i]);

10 Tries 1 / 4

TRIE IMPLEMENTATION

TrieNode + Trie class
class TrieNode {
    TrieNode[] children = new TrieNode[26];
    boolean isEnd = false;
}

class Trie {
    TrieNode root = new TrieNode();

    void insert(String word) {
        TrieNode node = root;
        for (char c : word.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null)
                node.children[i] = new TrieNode();
            node = node.children[i];
        }
        node.isEnd = true;
    }

    boolean search(String word) {
        TrieNode node = root;
        for (char c : word.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null) return false;
            node = node.children[i];
        }
        return node.isEnd;
    }

    boolean startsWith(String prefix) {
        TrieNode node = root;
        for (char c : prefix.toCharArray()) {
            int i = c - 'a';
            if (node.children[i] == null) return false;
            node = node.children[i];
        }
        return true;
    }
}

UNION-FIND (DISJOINT SET)

Template (path compression + union by rank)
int[] parent, rank;

void init(int n) {
    parent = new int[n];
    rank = new int[n];
    for (int i = 0; i < n; i++) parent[i] = i;
}

int find(int x) {
    if (parent[x] != x)
        parent[x] = find(parent[x]); // path compress
    return parent[x];
}

boolean union(int a, int b) {
    int ra = find(a), rb = find(b);
    if (ra == rb) return false; // already connected
    if (rank[ra] < rank[rb]) parent[ra] = rb;
    else if (rank[ra] > rank[rb]) parent[rb] = ra;
    else { parent[rb] = ra; rank[ra]++; }
    return true;
}
Union returns false = cycle detected. Use for: connected components, redundant connection.

TOPOLOGICAL SORT (CYCLE DETECT)

Course Schedule — BFS (Kahn's algorithm)
int[] inDegree = new int[n];
for (int[] pre : prerequisites)
    inDegree[pre[0]]++;

Queue<Integer> q = new LinkedList<>();

DIJKSTRA (SHORTEST PATH)

Weighted graph — min-heap by distance
int[] dist = new int[n];
Arrays.fill(dist, Integer.MAX_VALUE);
dist[src] = 0;

// {distance, node}
PriorityQueue<int[]> pq =
    new PriorityQueue<>((a,b) -> a[0] - b[0]);
pq.offer(new int[]{0, src});

while (!pq.isEmpty()) {
    int[] curr = pq.poll();
    int d = curr[0], node = curr[1];
    if (d > dist[node]) continue; // stale entry

    for (int[] neighbor : graph.get(node)) {
        int next = neighbor[0], w = neighbor[1];
        if (dist[node] + w < dist[next]) {
            dist[next] = dist[node] + w;
            pq.offer(new int[]{dist[next], next});
        }
    }
}
if (d > dist[node]) continue — this line is critical. Skips stale entries from heap.
